<script> (function() { "use strict"; // ============================================================ // УНИВЕРСАЛЬНЫЙ ОФЛАЙН-БЛОК // ============================================================ function registerSW() { if (!('serviceWorker' in navigator)) { console.warn('[SW] ❌ Service Worker не поддерживается'); return; } navigator.serviceWorker.getRegistration() .then((registration) => { if (registration) { console.log('[SW] ✅ Уже зарегистрирован:', registration); return; } navigator.serviceWorker.register('sw.js') .then((reg) => { console.log('[SW] ✅ Зарегистрирован:', reg); }) .catch((err) => { console.error('[SW] ❌ Ошибка:', err); }); }) .catch((err) => { console.error('[SW] ❌ Ошибка проверки:', err); }); } function checkNetwork() { const status = navigator.onLine ? 'Онлайн' : 'Офлайн (из кэша)'; console.log(`📡 ${status}`); } function setupNetworkListeners() { window.addEventListener('online', () => { console.log('📡 Сеть восстановлена'); if ('serviceWorker' in navigator) { navigator.serviceWorker.ready .then((registration) => { registration.update(); console.log('[SW] 🔄 Проверка обновлений...'); }) .catch(() => {}); } }); window.addEventListener('offline', () => { console.log('📡 Сеть отключена (работаем из кэша)'); }); } function init() { registerSW(); checkNetwork(); setupNetworkListeners(); console.log('[APP] ✅ Офлайн-функционал активирован'); } if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', init); } else { init(); } })(); </script>
